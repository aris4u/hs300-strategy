WASHOUT_TURN:=洗盘信号 AND NOT(REF(洗盘信号,1));
启动条件4:=COUNT(WASHOUT_TURN,10)>0;
LAUNCH_RAW:=洗盘结束启动;
LAUNCH_TURN:=LAUNCH_RAW AND NOT(REF(LAUNCH_RAW,1)) AND COUNT(LAUNCH_RAW AND NOT(REF(LAUNCH_RAW,1)),10)=1;
F_SIGNAL:=FZYJ AND NOT(REF(FZYJ,1));
RECENT_LAUNCH:=COUNT(LAUNCH_TURN,40)>0;
JC_CROSS:=JC_STATE AND CROSS(CHQD,50) AND RECENT_LAUNCH;
CT_CROSS:=CT_STATE AND CROSS(CHQD,70) AND RECENT_LAUNCH;
JC_EVENT:=JC_CROSS AND COUNT(JC_CROSS,10)=1;
ESCAPE_TOP:=CT_CROSS AND COUNT(CT_CROSS,10)=1;
REDUCE_TREND:=JC_EVENT AND QQS;
REDUCE_BAND:=JC_EVENT AND NOT(QQS);
CAUTION:=YJ_STATE AND CROSS(CHQD,25);

{----- 主图叠加：画在K线上 -----}
DRAWICON(LAUNCH_TURN,L*0.985,5);
DRAWTEXT(LAUNCH_TURN,L*0.97,'LAUNCH'),COLORYELLOW;

DRAWTEXT(F_SIGNAL,L*0.96,'F'),COLORMAGENTA;
DRAWICON(REDUCE_BAND,H*1.01,1);

DRAWICON(REDUCE_TREND,H*1.01,1);

DRAWICON(ESCAPE_TOP,H*1.015,2);
DRAWTEXT(ESCAPE_TOP,H*1.03,'CT'),COLORMAGENTA;
DRAWTEXT(CAUTION,H,'YJ'),COLORCYAN;
